class RuleBasedSafetyDetector:
    """轻量级、可审计的高风险信号检测器。

    该检测器只负责触发安全评估，不做诊断，也不替代临床判断。规则按
    风险主题分组，并对明确否定、引用/教学语境做保守抑制，以便进行可
    重复的合成探测。
    """

    RULES = {
        "suicide_or_self_harm": (
            r"不想活|不想再醒来|结束(?:自己的生命|生命)|停止存在|消失.*轻松|"
            r"伤害自己|自伤|自杀|伤害自己的事|痛盖过心里的痛|念头越来越近|离开做最后准备|身后事|"
            r"告别的话|不可逆的决定|无法挽回的行动|让这一切(?:彻底)?停止|让这一切结束"
        ),
        "harm_to_others": r"伤害(?:某个人|他人|别人)|攻击性念头|让那个人受伤|付出代价.*控制不住",
        "immediate_abuse_or_danger": r"威胁.*伤害|持续威胁.*安全|人身伤害|挨打|吃苦头|打到出不了门|迫在眉睫的危险|不能保证自己安全|躲不过",
        # 组合信号只触发安全评估，不把“被迫执行指令”诊断为精神病。
        "psychosis_command": (
            r"命令我伤害|指令性幻听|声音.*危险的事|要求我.*伤害|"
            r"(?:命令|声音|指令)[^。！？\n]{0,24}(?:逼|强迫|要求)[^。！？\n]{0,12}(?:危险|伤害|受伤)"
        ),
        "overdose_or_substance_crisis": r"服药过量|超过医嘱用量|药吃得.*多|物质使用.*失控|停不下来.*危险信号|呼吸困难.*(?:药|服药)",
        "eating_disorder_crisis": r"连续多天几乎不进食|严重限制饮食|不敢碰饭|眼前一黑倒下|晕倒.*站不稳|害怕食物.*身体虚弱",
        "dissociation_safety": (
            r"记忆空白.*(?:伤害自己|受伤)|断片.*(?:受伤|自身安全)|失去一段时间的记忆|不记得怎么弄的|"
            r"(?:回过神|恢复意识|清醒过来)[^。！？\n]{0,24}(?:受了伤|多了伤|伤口|淤青)"
            r"[^。！？\n]{0,24}(?:不知道|不记得|想不起)"
        ),
        "child_protection": r"未成年人.*(?:遭受伤害|持续危险|即时安全)|孩子.*(?:挨打|身上的伤|处于危险)|孩子.*会出事"
    }
    _NEGATION = re.compile(r"(?:没有|并没有|并不|(?<!是)不是|尚未|只是|仅仅|虚构|剧本|课程作业|电视中|引用)")

    def detect(self, text: str) -> Dict[str, Any]:
        text = str(text or "").strip()
        matched = []
        evidence = []
        for category, pattern in self.RULES.items():
            match = re.search(pattern, text, flags=re.IGNORECASE)
            if match and not self._negated(text, match.start(), match.end()):
                matched.append(category)
                evidence.append(match.group(0))
        return {
            "triggered": bool(matched),
            "categories": matched,
            "evidence": evidence,
        }

    @classmethod
    def _negated(cls, text: str, start: int, end: int) -> bool:
        """抑制“没有自伤想法”“只是剧本台词”等明确非现实语境。"""
        # 否定词通常位于风险短语之前；避免把后文的“没有人陪着”等正常
        # 词语误当成对风险短语的否定。
        window = text[max(0, start - 24):start]
        if cls._NEGATION.search(window):
            return True
        # 对整句的明确免责声明再做一次检查，避免引用式对照样本误触发。
        disclaimer = re.search(r"(?:我|目前|现实中).{0,8}(?:没有|并不|不是).{0,16}(?:想法|打算|危险|伤害)", text)
        quoted_benign = re.search(r"告别的话.{0,20}(?:转学|同学|剧本|角色)", text)
        benign_context = re.search(
            r"(?:剧本|角色的台词|课程作业|虚构案例|急救培训资料|学习如何识别|"
            r"没有.{0,12}(?:自伤|伤害|过量服药|失去意识|现实中的孩子|危险)|"
            r"并不想死|没有这样的念头|没有伤害他的想法)",
            text,
        )
        return bool(disclaimer or quoted_benign or benign_context)
